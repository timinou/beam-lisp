(ns research.constructor.declaration-kind)
;; Compiling self-quoted syntax must classify the macro from Source metadata,
;; never recursively compile or execute its initializer/body for the boolean.
(defmacro self-quoted [] `(quote self-quoted))
;; Inspecting this macro while qualifying a quote must not execute its body.
(defmacro inspected [] (erlang/error :macro-kind-inspection-executed-body))
(defmacro quotes-inspected [] `(quote inspected))
(def self-result (self-quoted))
(def inspected-result (quotes-inspected))
