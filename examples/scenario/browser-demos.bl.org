#+TITLE: Two browser demos, two ordinary tests

Each demo keeps its actions, assertions and client presentation together.
Run this file with =bl test= for capture, or set =BL_DEMO_EXPORT=1= to also
create narrated videos. =bl run= executes the same registered tests.

Add =optional/browser-port/bl= to the source path. Set =BL_BROWSER_PORT=,
=CHROMIUM=, =SWAY= and =WAYVNC= for the optional native browser runtime.
Speech export reads =DEEPGRAM_API_KEY= from the environment or ignored =.env=.

* Shared browser setup

#+begin_src beam-lisp
(ns examples.browser-demos
  (:require [scenario :as s] [scenario.demo :as demo] [browser.film :as browser]
            [browser.page :as page]))

(def output (or (System/get_env "BL_DEMO_OUTPUT") "!tasks/evidence/browser-demos"))
(defn options [id]
  {:output (str output "/" id)
   :film {:voice "aura-2-hector-fr" :language "fr" :label "Dossier de démonstration"}
   :export {:env-file ".env"}
   :world (fn [{:keys [frames]}]
            (browser/owned-world :viewer
              {:width 1280 :height 800
               :launch {"chromium" (System/get_env "CHROMIUM")
                        "compositor" (System/get_env "SWAY")
                        "vnc" (System/get_env "WAYVNC") "width" 1280 "height" 900}}
              frames))})
#+end_src

* Read a message

Scene annotations wrap the usual scenario events. The click is captured from
real browser input; its position drives the pointer in the exported film.

#+begin_src beam-lisp
(demo/defdemo read-a-message (options "read-a-message")
  (s/scenario "Consulter un message" {:viewer {}}
    [(s/does :viewer [:navigate "data:text/html,<h1>Votre messagerie</h1><button id='message' onclick='this.textContent=\"Message lu\"'>Nouveau message</button>"])
     (s/does :viewer [:wait "!!document.querySelector('#message')"])
     (demo/scene :read {:title "Consulter vos messages"
                       :say "Ouvrez votre message. Vous voyez immédiatement qu’il a été lu."
                       :camera [{:at 0 :zoom 1} {:at 2 :zoom 1.4 :x 0.3 :y 0.2}]}
       (s/does :viewer [:click "#message"])
       (s/sees :viewer (fn [p _]
         (= "Message lu" (page/evaluate! p "document.querySelector('#message').textContent")))
         "The message is marked as read"))]))
#+end_src

* Personalize your name

This is an independent test with a fresh browser, its own narration and its
own exported movie. Shared helpers remain ordinary Beam Lisp functions.

#+begin_src beam-lisp
(demo/defdemo personalize-your-name (options "personalize-your-name")
  (s/scenario "Personnaliser votre profil" {:viewer {}}
    [(s/does :viewer [:navigate "data:text/html,<h1>Votre profil</h1><input id='name'><button id='save' onclick='document.querySelector(\"h1\").textContent=\"Bonjour, \"+document.querySelector(\"input\").value'>Enregistrer</button>"])
     (s/does :viewer [:wait "!!document.querySelector('#name')"])
     (demo/scene :name {:title "Personnaliser votre profil"
                       :say "Choisissez votre nom et enregistrez. Votre profil est à jour."
                       :subtitle "Choisissez votre nom et enregistrez. Votre profil est à jour."}
       (s/does :viewer [:fill "#name" "Camille"])
       (s/does :viewer [:click "#save"])
       (s/sees :viewer (fn [p _]
         (= "Bonjour, Camille" (page/evaluate! p "document.querySelector('h1').textContent")))
         "The saved name appears in the greeting"))]))
